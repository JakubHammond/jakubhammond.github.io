# ALU 1001

This SystemVerilog code is already tested and synthesized and correct. I named the file alu.sv. It is a simple ALU that can do standard operations such as addition, subtraction, AND operation, OR operation, XOR operation, Shift Left operation, Shift Right operation:

```systemverilog
`timescale 1ns/1ps

module alu #(
    parameter WIDTH = 8
)(
    input  logic [WIDTH-1:0] a,
    input  logic [WIDTH-1:0] b,
    input  logic [2:0]       op,
    output logic [WIDTH-1:0] result,
    output logic             zero,
    output logic             carry
);

    localparam ADD = 3'b000;
    localparam SUB = 3'b001;
    localparam AND = 3'b010;
    localparam OR  = 3'b011;
    localparam XOR = 3'b100;
    localparam SL  = 3'b101;
    localparam SR  = 3'b110;

    logic [WIDTH:0] tempresult;

    always_comb begin
        carry  = 0;
        zero   = 0;
        result = '0;
        tempresult = '0;

        case(op)
            ADD: begin
                tempresult = {1'b0, a} + {1'b0, b};
                carry = tempresult[WIDTH];
            end
            SUB: begin
                tempresult = {1'b0, a} - {1'b0, b};
                carry = tempresult[WIDTH];
            end
            AND: tempresult = {1'b0, a & b};
            OR:  tempresult = {1'b0, a | b};
            XOR: tempresult = {1'b0, a ^ b};
            SL: tempresult = {a, 1'b0};
            SR: tempresult = ({1'b0, a} >> 1);
            default: tempresult = '0;
        endcase

        result = tempresult[WIDTH-1:0];
        zero   = ~|result;
    end

endmodule
```

This is the test-bench code I named the file alu_tb.sv:
```systemverilog
`timescale 1ns/1ps

module alu_tb;

    parameter WIDTH = 8;

    logic [WIDTH-1:0] a, b;
    logic [2:0]       op;
    logic [WIDTH-1:0] result;
    logic             zero, carry;

    alu #(.WIDTH(WIDTH)) dut (
        .a(a),
        .b(b),
        .op(op),
        .result(result),
        .zero(zero),
        .carry(carry)
    );

    initial begin
        $display("ALU Test Start");

        for (int k = 0; k <= 6; k++) begin
            for (int i = 0; i <= 3; i++) begin
                for (int j = 0; j <= 3; j++) begin
                    op = k[2:0];       // force 3 bits
                    a  = i[WIDTH-1:0]; // force WIDTH bits
                    b  = j[WIDTH-1:0];
                    #1;
                    $display("op=%0d a=%0d b=%0d => result=%0d carry=%0b zero=%0b",
                             op, a, b, result, carry, zero);
                end
            end
        end

        $finish;
    end

endmodule
```
This is the simulation C++ code, the file is named sim_main.cpp:
```cpp
#include "Valu_tb.h"
#include "verilated.h"

int main(int argc, char **argv, char **env) {
    Verilated::commandArgs(argc, argv);
    Valu_tb* top = new Valu_tb;

    while (!Verilated::gotFinish()) {
        top->eval();
        Verilated::timeInc(1);  // optional, for accurate # delays
    }

    delete top;
    return 0;
}
```
